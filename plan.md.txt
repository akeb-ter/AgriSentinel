Python
import cv2
import numpy as np
from tflite_runtime.interpreter import Interpreter

class PestDetector:
    def __init__(
        self,
        model_path: str = "edge/models/pest_detector.tflite",
        conf_threshold: float = 0.60,
        iou_threshold: float = 0.45
    ):
        self.conf_threshold = conf_threshold
        self.iou_threshold = iou_threshold

        # Initialize TFLite interpreter and allocate memory buffers
        self.interpreter = Interpreter(model_path=model_path)
        self.interpreter.allocate_tensors()

        self.input_details = self.interpreter.get_input_details()
        self.output_details = self.interpreter.get_output_details()

        # Input dimensions (expected: 416x416)
        self.input_height = self.input_details[0]['shape'][1]
        self.input_width = self.input_details[0]['shape'][2]

        # Target IP102 Class IDs for Rice, Corn, Eggplant, Squash, and Mungbean pests
        self.ALLOWED_CLASSES = {
            0: "Rice leaf roller",
            1: "Rice leaf caterpillar",
            3: "Asiatic rice borer",
            4: "Yellow rice borer",
            7: "Brown plant hopper",
            8: "White backed plant hopper",
            12: "Grain spreader thrips",
            18: "Black cutworm",
            19: "Large cutworm",
            45: "Corn leaf aphid",
            56: "Fall armyworm",
            78: "Whitefly"
        }

    def preprocess(self, frame: np.ndarray) -> np.ndarray:
        """Resizes, normalizes, and expands dimensions for model inference."""
        resized = cv2.resize(frame, (self.input_width, self.input_height))
        rgb = cv2.cvtColor(resized, cv2.COLOR_BGR2RGB)
        normalized = rgb.astype(np.float32) / 255.0
        return np.expand_dims(normalized, axis=0)

    def detect(self, frame: np.ndarray) -> list[dict]:
        """Runs inference, filters for target crop pests, and applies NMS."""
        orig_h, orig_w = frame.shape[:2]
        input_data = self.preprocess(frame)

        self.interpreter.set_tensor(self.input_details[0]['index'], input_data)
        self.interpreter.invoke()

        # YOLO TFLite output shape: [1, 4 + classes, anchors] -> Transpose to [anchors, 4 + classes]
        raw_output = self.interpreter.get_tensor(self.output_details[0]['index'])
        predictions = np.squeeze(raw_output).T

        boxes = []
        scores = []
        class_ids = []

        for pred in predictions:
            class_scores = pred[4:]
            max_idx = int(np.argmax(class_scores))
            confidence = float(class_scores[max_idx])

            # Apply confidence threshold and isolate allowed target crop pests
            if confidence >= self.conf_threshold and max_idx in self.ALLOWED_CLASSES:
                cx, cy, w, h = pred[0:4]
                x1 = int((cx - w / 2) * orig_w)
                y1 = int((cy - h / 2) * orig_h)
                box_w = int(w * orig_w)
                box_h = int(h * orig_h)

                boxes.append([x1, y1, box_w, box_h])
                scores.append(confidence)
                class_ids.append(max_idx)

        # Non-Maximum Suppression to remove redundant boxes
        indices = cv2.dnn.NMSBoxes(boxes, scores, self.conf_threshold, self.iou_threshold)

        detected_pests = []
        if len(indices) > 0:
            for idx in indices.flatten():
                detected_pests.append({
                    "class_id": class_ids[idx],
                    "name": self.ALLOWED_CLASSES[class_ids[idx]],
                    "confidence": round(scores[idx], 4),
                    "box": boxes[idx]
                })

        return detected_pests
